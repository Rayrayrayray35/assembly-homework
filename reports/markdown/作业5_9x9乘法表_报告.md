## 三、汇编实现

### 3.1 双重循环

- 外层用 **BX** 保存行号，从 9 递减到 1：`MOV BX,9` … `DEC BX` / `JNZ ROW_LOOP`。
- 内层用 **CX** 保存列号，从 1 递增到当前行号：`MOV CX,1` … `INC CX` / `CMP CX,BX` / `JBE`。
- 每行最后一个式子是 `row*row`，后面不再输出空格，直接回车换行：

  ```asm
  CMP CX,BX
  JAE END_ROW        ; col >= row，本行结束
  MOV DL,' '
  CALL PRINT_CHAR
  INC CX
  JMP COL_LOOP
  ```

### 3.2 过程调用与寄存器保护

程序定义了三个过程，全部通过 `CALL` 调用、`RET` 返回：

| 过程 | 作用 | 入口保存 | 出口恢复 |
|------|------|----------|----------|
| `PRINT_DEC` | 把 `AX` 按十进制输出 | `AX/BX/CX/DX` 全部 `PUSH` | 逆序 `POP` |
| `PRINT_CHAR` | 输出 `DL` 中的字符 | `AX/DX` | 逆序 `POP` |
| `NEWLINE` | 输出 CR LF | `AX/DX` | 逆序 `POP` |

关键点：`PRINT_DEC` 内部要用 `DIV BX`（会用到 `AX/DX`）、要用 `CX` 数位数、要用 `BX` 放除数 10；而主过程正用 `BX` 存行号、`CX` 存列号。因此过程入口必须把 `AX/BX/CX/DX` 全部压栈，出口再按“后进先出”的顺序恢复。否则返回主过程后，`BX` 或 `CX` 被改掉，双重循环就会错乱。

```asm
PRINT_DEC PROC
        PUSH AX
        PUSH BX
        PUSH CX
        PUSH DX
        ...
        POP DX
        POP CX
        POP BX
        POP AX
        RET
PRINT_DEC ENDP
```

`PRINT_CHAR` 也不能只保存 `DX`：`INT 21H` 的 02H 功能要求功能号放在 `AH`，会覆盖 `AX` 的高字节，所以 `AX` 也要保存。

### 3.3 乘法与十进制输出

乘法用 `MUL`：

```asm
MOV AX,BX       ; AX = row
MUL CX          ; DX:AX = AX * CX
CALL PRINT_DEC  ; 输出乘积
```

本题最大乘积是 `9*9=81`，结果不超过 255，`DX` 必为 0；但 `MUL` 的完整结果在 `DX:AX`，写法上仍按 16 位结果处理。

十进制输出用“除 10 取余、余数压栈、再逆序弹出”：

```asm
MOV CX,0
MOV BX,10
PD_DIV:
        MOV DX,0
        DIV BX
        PUSH DX
        INC CX
        CMP AX,0
        JNE PD_DIV
PD_OUT:
        POP DX
        ADD DL,'0'
        MOV AH,02H
        INT 21H
        LOOP PD_OUT
```

例如 `81`：先得到余数 1、8，压栈后弹出顺序为 8、1，正好输出 `81`。

### 3.4 完整汇编源码

```asm
;===========================================================
;  MUL9.ASM -- 九九乘法表（9×9）
;  题目: 作业题目4 · 乘法指令和过程调用
;===========================================================
        .MODEL SMALL
        .STACK 100H

        .DATA
MSG     DB  'The 9mul9 table:',0DH,0AH,'$'

        .CODE
START:  MOV AX,@DATA
        MOV DS,AX

        LEA DX,MSG              ; 显示表头
        MOV AH,09H
        INT 21H

        MOV BX,9                ; BX = 外层行号（9..1）

ROW_LOOP:
        MOV CX,1                ; CX = 内层列号（1..BX）

COL_LOOP:
        MOV AX,BX               ; 输出 row
        CALL PRINT_DEC

        MOV DL,'*'              ; 输出 '*'
        CALL PRINT_CHAR

        MOV AX,CX               ; 输出 col
        CALL PRINT_DEC

        MOV DL,'='              ; 输出 '='
        CALL PRINT_CHAR

        MOV AX,BX               ; AX = row
        MUL CX                  ; AX = row * col（结果 <= 81，DX=0）
        CALL PRINT_DEC          ; 输出乘积

        CMP CX,BX               ; 该行最后一个？
        JAE END_ROW             ; col >= row -> 换行
        MOV DL,' '              ; 否则输出一个空格
        CALL PRINT_CHAR
        INC CX
        JMP COL_LOOP

END_ROW:
        CALL NEWLINE            ; 一行结束，回车换行
        DEC BX                  ; 下一行（行号减 1）
        JNZ ROW_LOOP            ; BX != 0 继续

        MOV AX,4C00H            ; 返回 DOS
        INT 21H

;-----------------------------------------------------------
; PRINT_DEC: 把 AX 按无符号十进制输出
; 过程中使用 AX/BX/CX/DX，所以入口全部 PUSH，出口全部 POP，
; 保证主过程的 BX（行号）、CX（列号）不被破坏。
;-----------------------------------------------------------
PRINT_DEC PROC
        PUSH AX
        PUSH BX
        PUSH CX
        PUSH DX

        MOV CX,0                ; CX = 位数
        MOV BX,10               ; 除数为 10
PD_DIV:
        MOV DX,0                ; DX:AX / 10
        DIV BX
        PUSH DX                 ; 余数压栈
        INC CX
        CMP AX,0
        JNE PD_DIV

PD_OUT:
        POP DX                  ; 逆序弹出：高位先输出
        ADD DL,'0'
        MOV AH,02H
        INT 21H
        LOOP PD_OUT

        POP DX
        POP CX
        POP BX
        POP AX
        RET
PRINT_DEC ENDP

;-----------------------------------------------------------
; PRINT_CHAR: 输出 DL 中的一个字符
; INT 21H 的 02H 功能会用到 AH，因此保存/恢复 AX。
;-----------------------------------------------------------
PRINT_CHAR PROC
        PUSH AX
        PUSH DX
        MOV AH,02H
        INT 21H
        POP DX
        POP AX
        RET
PRINT_CHAR ENDP

;-----------------------------------------------------------
; NEWLINE: 输出 CR LF
;-----------------------------------------------------------
NEWLINE PROC
        PUSH AX
        PUSH DX
        MOV DL,0DH
        MOV AH,02H
        INT 21H
        MOV DL,0AH
        MOV AH,02H
        INT 21H
        POP DX
        POP AX
        RET
NEWLINE ENDP

END START
```

## 四、C 语言实现

```c
#include <stdio.h>

int main(void)
{
    int i, j;

    printf("The 9mul9 table:\n");
    for (i = 9; i >= 1; i--) {
        for (j = 1; j <= i; j++) {
            printf("%d*%d=%d", i, j, i * j);
            if (j < i) {
                putchar(' ');
            }
        }
        putchar('\n');
    }
    return 0;
}
```

编译并运行：

```dos
TCC -IC:\TURBOC3\INCLUDE -LC:\TURBOC3\LIB -ms MUL9C.C
MUL9C
```

输出与汇编版完全一致。

## 五、反汇编与注释

用 `TCC -ms -S MUL9C.C` 生成 `MUL9C.ASM`。下面只保留 `main` 的关键部分，完整文件见 `src/05_mul9/mul9c_annotated.asm`。

```asm
_main   proc near
        push    bp              ; 保存调用者的 BP
        mov     bp,sp           ; 建立当前栈帧
        push    si              ; 保存 SI（i）
        push    di              ; 保存 DI（j）

;       printf("The 9mul9 table:\n");
        mov     ax,offset DGROUP:s@
        push    ax
        call    near ptr _printf
        pop     cx              ; cdecl：调用者清理参数

;       for (i = 9; i >= 1; i--)
        mov     si,9            ; i 放在 SI
        jmp     short @1@366

@1@58:
        mov     di,1            ; j 放在 DI
        jmp     short @1@226

@1@86:
;       printf("%d*%d=%d", i, j, i * j)
        mov     ax,si           ; AX = i
        imul    di              ; DX:AX = i * j（有符号乘法）
        push    ax              ; 第 4 个参数：乘积
        push    di              ; 第 3 个参数：j
        push    si              ; 第 2 个参数：i
        mov     ax,offset DGROUP:s@+18
        push    ax              ; 第 1 个参数：格式串
        call    near ptr _printf
        add     sp,8            ; 4 个 word 参数 = 8 字节

;       if (j < i) putchar(' ');
        cmp     di,si
        jge     short @1@198
        ; （此处为 Turbo C 对 putchar 的内联优化，省略）
@1@198:
        inc     di              ; j++
@1@226:
        cmp     di,si
        jle     short @1@86     ; j <= i 继续内层

;       putchar('\n')
        ; （同样被内联为对 __streams 缓冲区的操作，省略）
@1@338:
        dec     si              ; i--
@1@366:
        cmp     si,1
        jge     short @1@58     ; i >= 1 继续外层

        xor     ax,ax           ; return 0
@1@422:
        pop     di              ; 恢复 DI
        pop     si              ; 恢复 SI
        pop     bp              ; 恢复 BP
        ret
_main   endp
```

### 5.1 对照分析

1. **循环变量分配**：Turbo C 把外层 `i` 放在 `SI`，内层 `j` 放在 `DI`；因为 `SI/DI` 是调用者保存寄存器，函数入口先 `PUSH SI`、`PUSH DI`，返回前再 `POP DI`、`POP SI`。这与我们汇编版用 `BX/CX` 并手动保护是同一个道理。
2. **乘法指令**：C 的 `i * j` 编译成 `MOV AX,SI` + `IMUL DI`。`IMUL` 是有符号乘法，结果在 `DX:AX`；本题数值很小，`DX=0`。汇编版用 `MUL CX`（无符号），因为行列号都是非负的。
3. **参数传递**：`printf` 是可变参数函数，采用 cdecl 调用约定，参数从右向左压栈；调用后由调用者 `ADD SP,8` 清理 4 个 word 参数。汇编版没有调用 C 库，而是自己用 `INT 21H` 输出，所以不需要这部分。
4. **`putchar` 的内联**：Turbo C 把 `putchar(' ')` 和 `putchar('\n')` 优化成直接操作 `__streams` 缓冲区（`AL=32` 或 `10`），没有真正调用 `__fputc`；只有缓冲区满时才走 `__fputc`。这说明“库函数”在汇编层不一定表现为一次 `CALL`。
5. **栈帧**：`PUSH BP` / `MOV BP,SP` 建立栈帧，`PUSH SI` / `PUSH DI` 保存寄存器；返回前逆序恢复。过程调用的“现场保护”在编译器生成代码里同样清晰可见。

## 七、小结

1. **过程调用的本质是栈**：`CALL` 把返回地址压栈，`RET` 弹出返回地址；过程内部用 `PUSH/POP` 保护自己会修改、但主过程还要继续使用的寄存器。
2. **寄存器保护要成对**：入口 `PUSH` 的顺序与出口 `POP` 的顺序相反（后进先出）。`PRINT_DEC` 如果漏掉 `BX` 或 `CX`，外层行号或内层列号就会被除法和计数破坏，输出立刻错乱。
3. **乘法与除法要关注寄存器对**：`MUL` 的结果在 `DX:AX`，`DIV` 的被除数也在 `DX:AX`；16 位除法前必须把 `DX` 清零，否则会溢出。
4. **C 与汇编可以互相对照**：C 的 `for`、`i*j`、`printf` 在汇编层分别对应比较/条件跳转、`IMUL`、压栈传参和 `CALL`；编译器的寄存器分配和调用约定，正好印证了手写汇编时的现场保护问题。
